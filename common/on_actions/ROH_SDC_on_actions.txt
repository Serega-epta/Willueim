on_actions = {
	on_daily_SDC = {
		effect ={
			if = {
				limit = {
					check_variable = {
						num_divisions > 0
					}
				}
				set_variable = {
					sdc_division_counter = {
						value = num_divisions
						divide = {
							value = num_divisions
							add = VOZ.num_divisions
						}
					}
				}
			}
			else = {
				set_variable = {
					sdc_division_counter = 0
				}
			}
		}
	}
}
