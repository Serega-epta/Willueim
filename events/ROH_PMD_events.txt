country_event = {
	id = PMD_expedition.10
	title = PMD_expedition.10.t
	desc = PMD_expedition.10.d
	fire_only_once = yes
	is_triggered_only = yes
	option = {
		name = PMD_expedition.10.a
		IF = {
			limit = {
				has_country_flag = expedition_1_flag
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = 500
			}
			add_equipment_to_stockpile = {
				type = support_equipment
				amount = 100
			}
			add_equipment_to_stockpile = {
				type = civilian_clothing
				amount = 200
			}
		}
		else_if = {
			limit = {
				has_country_flag = expedition_2_flag
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = 600
			}
			add_equipment_to_stockpile = {
				type = support_equipment
				amount = 150
			}
			add_equipment_to_stockpile = {
				type = civilian_clothing
				amount = 300
			}
		}
		else_if = {
			limit = {
				has_country_flag = expedition_3_flag
			}

				type = infantry_equipment
				amount = 1000
			}
